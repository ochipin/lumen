Do not call .Paginate here: a list that did not opt in must not generate pager pages or even Hugo's /page/1/ alias. */ -}} {{- $total := len $entries -}} {{- $pager = dict "Pages" $entries "PageNumber" 1 "PagerSize" (math.Max 1 $total | int) "TotalNumberOfElements" $total "NumberOfElements" $total "TotalPages" 1 "URL" .RelPermalink "Pagers" (slice (dict "URL" .RelPermalink)) "HasPrev" false "HasNext" false "Prev" false "Next" false -}} {{- end -}} {{- return $pager -}}
