{{- /* RenderString does not expose whether Hugo's id was explicit or automatic. Keep the existing scoped heading ids, and expose each original id only when it is unambiguous across the finished page. This preserves authored bookmarks without breaking old scoped links or changing normal Markdown heading ids. */ -}} {{- $html := string . -}} {{- if not (strings.Contains $html " data-lumen-original-id=") }}{{ return ($html | safeHTML) }}{{ end -}} {{- /* Examples and inert HTML must not participate in bookmark allocation. */ -}} {{- $masks := slice -}} {{- $prefix := printf "LUMENHEADINGMASK%s" (crypto.SHA256 $html) -}} {{- $patterns := slice `(?s)|$)` -}} {{- range slice "script" "style" "pre" "template" "textarea" -}} {{- $patterns = $patterns | append (printf `(?is)<%s\b[^>]*>.*?(?:|$)` . .) -}} {{- end -}} {{- range findRE (delimit $patterns "|") $html -}} {{- $token := printf "%s%dEND" $prefix (len $masks) -}} {{- $masks = $masks | append (dict "token" $token "html" .) -}} {{- $html = replace $html . $token 1 -}} {{- end -}} {{- $used := newScratch -}} {{- $counts := newScratch -}} {{- $headings := slice -}} {{- $attributes := `(?s)\s+([^\s"'=<>/]+)(?:\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s"'=<>]+)))?` -}} {{- range findRE `<[A-Za-z][^>"']*(?:(?:"[^"]*"|'[^']*')[^>"']*)*>` $html -}} {{- $tag := . -}} {{- $heading := findRE `(?i)^)` $tag 1 -}} {{- $anchor := findRE `(?i)^)` $tag 1 -}} {{- $original := "" -}} {{- $marker := "" -}} {{- range findRESubmatch $attributes $tag -}} {{- $name := lower (index . 1) -}} {{- $value := htmlUnescape ((index . 2) | default (index . 3) | default (index . 4)) -}} {{- if or (eq $name "id") (and $anchor (eq $name "name")) -}} {{- if $value }}{{ $used.Set $value true }}{{ end -}} {{- else if and $heading (eq $name "data-lumen-original-id") -}} {{- $original = $value -}} {{- $marker = index . 0 -}} {{- end -}} {{- end -}} {{- if $marker -}} {{- $headings = $headings | append (dict "tag" $tag "marker" $marker "original" $original) -}} {{- if $original }}{{ $counts.Add $original 1 }}{{ end -}} {{- end -}} {{- end -}} {{- range $headings -}} {{- $tag := replace .tag .marker "" 1 -}} {{- if and .original (eq ($counts.Get .original) 1) (not ($used.Get .original)) -}} {{- $tag = printf `%s` $tag (htmlEscape .original) -}} {{- $used.Set .original true -}} {{- end -}} {{- $html = replace $html .tag $tag 1 -}} {{- end -}} {{- range $masks }}{{ $html = replace $html .token .html 1 }}{{ end -}} {{- return ($html | safeHTML) -}}
